#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""todo.py — 密闭 workspace 内的 CLI 待办事项小工具（造物期·第12轮·最小重写版）
仅标准库，全相对路径，无网络。命令：add / list / done / del
"""
import argparse
import json
import os
import sys

DATA_FILE = os.path.join(os.path.dirname(os.path.abspath(__file__)), "todo.json")

P1_4_HELP_TEXT = """用法: python todo.py <命令> [参数]
命令:
  add <标题>    新增待办
  list          列出全部待办
  done <序号>   完成指定待办
  del <序号>    删除指定待办
提示: 不带参数运行将显示本帮助。
"""


def load_tasks():
    """修补点③：todo.json 损坏时自动改名 .bak 并重建空结构"""
    if not os.path.exists(DATA_FILE):
        return []
    try:
        with open(DATA_FILE, "r", encoding="utf-8") as f:
            data = json.load(f)
        if not isinstance(data, dict) or not isinstance(data.get("tasks"), list):
            raise ValueError("结构不符")
        return data["tasks"]
    except (json.JSONDecodeError, ValueError, OSError) as e:
        bak = DATA_FILE + ".bak"
        try:
            if os.path.exists(bak):
                os.remove(bak)
            os.rename(DATA_FILE, bak)
        except OSError:
            pass
        print("[恢复] todo.json 已损坏(%s)，已备份为 todo.json.bak 并重建空列表" % e)
        return []


def save_tasks(tasks):
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump({"tasks": tasks}, f, ensure_ascii=False, indent=2)


def _parse_index(raw, tasks, action):
    """修补点②：done/del 非法序号明确报错，无 traceback"""
    try:
        idx = int(raw)
    except (TypeError, ValueError):
        print("错误: 序号必须是数字，收到 '%s'" % raw)
        return None
    if idx < 1 or idx > len(tasks):
        print("错误: 序号 %d 越界（有效范围 1~%d）" % (idx, max(len(tasks), 1)))
        return None
    return idx - 1


def cmd_add(args):
    tasks = load_tasks()
    tasks.append({"id": len(tasks) + 1, "title": args.title, "done": False})
    save_tasks(tasks)
    print("已添加: %s" % args.title)


def cmd_list(args):
    tasks = load_tasks()
    if not tasks:  # 修补点①：空列表友好提示
        print("（暂无待办。用 add 命令添加第一条吧）")
        return
    for i, t in enumerate(tasks, 1):
        mark = "x" if t.get("done") else " "
        print("[%d] [%s] %s" % (i, mark, t.get("title", "")))


def cmd_done(args):
    tasks = load_tasks()
    idx = _parse_index(args.index, tasks, "done")
    if idx is None:
        sys.exit(1)
    tasks[idx]["done"] = True
    save_tasks(tasks)
    print("已完成: %s" % tasks[idx]["title"])


def cmd_del(args):
    tasks = load_tasks()
    idx = _parse_index(args.index, tasks, "del")
    if idx is None:
        sys.exit(1)
    print("已删除: %s" % tasks.pop(idx)["title"])
    save_tasks(tasks)


def main(argv=None):
    parser = argparse.ArgumentParser(
        prog="todo.py", description="密闭 workspace 待办 CLI",
        formatter_class=argparse.RawDescriptionHelpFormatter, epilog=P1_4_HELP_TEXT)
    sub = parser.add_subparsers(dest="command")
    p_add = sub.add_parser("add", help="新增待办")
    p_add.add_argument("title")
    sub.add_parser("list", help="列出待办")
    p_done = sub.add_parser("done", help="完成待办")
    p_done.add_argument("index")
    p_del = sub.add_parser("del", help="删除待办")
    p_del.add_argument("index")
    args = parser.parse_args(argv)
    if args.command is None:  # 修补点④：无参数输出帮助
        parser.print_help()
        return 0
    return {"add": cmd_add, "list": cmd_list,
            "done": cmd_done, "del": cmd_del}[args.command](args) or 0


if __name__ == "__main__":
    sys.exit(main())
